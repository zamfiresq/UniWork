-------------------CREATE TABLE-----------------------

--PACIENT

CREATE SEQUENCE PACIENT_COD_PACIENT_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE PACIENT(
                    	COD_PACIENT INT
                        	CONSTRAINT PACIENT_COD_PACIENT_PK PRIMARY KEY,
                    	NUME VARCHAR2(25)
                        	CONSTRAINT PACIENT_NUME_NN NOT NULL,
                    	PRENUME VARCHAR2(30)
                        	CONSTRAINT PACIENT_PRENUME_NN NOT NULL,
                    	SEX VARCHAR2(10)
                        	CONSTRAINT PACIENT_SEX_NN NOT NULL,
                    	CNP VARCHAR2(20)
                        	CONSTRAINT PACIENT_CNP_NN NOT NULL,
                    	NR_TELEFON VARCHAR2(15),
                    	ADRESA VARCHAR2(40)
);


--CATEGORIE

CREATE SEQUENCE CATEGORIE_ID_CATEGORIE_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE CATEGORIE(
	ID_CATEGORIE INT
                	CONSTRAINT CATEGORIE_ID_CATEGORIE_PK PRIMARY KEY,
	DENUMIRE VARCHAR2(30)
                	CONSTRAINT CATEGORIE_DENUMIRE_NN NOT NULL
);


--PERSONAL MILITAR

CREATE SEQUENCE PERS_MIL_ID_MILITAR_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE PERS_MIL(
	ID_MILITAR INT
                	CONSTRAINT PERS_MIL_ID_MILITAR_PK PRIMARY KEY,
	ID_CATEGORIE INT
                	CONSTRAINT PERS_MIL_ID_CATEGORIE_FK REFERENCES CATEGORIE(ID_CATEGORIE) ON DELETE CASCADE,
	NUME VARCHAR2(25)
                	CONSTRAINT PERS_MIL_NUME_NN NOT NULL,
	PRENUME VARCHAR2(30)
                	CONSTRAINT PERS_MIL_PRENUME_NN NOT NULL,
	GRAD_MILITAR VARCHAR2(15)
                         	CONSTRAINT PERS_MIL_GRAD_MILITAR_NN NOT NULL,
	SALARIU NUMBER(8,2)
                         	CONSTRAINT PERS_MIL_SALARIU_CK CHECK ( SALARIU >= 5000 ),
	DATA_ANGAJARE DATE,
	EMAIL VARCHAR2(25)
);


--PROGRAMARE

CREATE SEQUENCE PROGRAMARE_ID_PROGRAMARE_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE PROGRAMARE(
	ID_PROGRAMARE INT
                	CONSTRAINT PROGRAMARE_ID_PROGRAMARE_PK PRIMARY KEY,
	COD_PACIENT INT
                	CONSTRAINT PROGRAMARE_COD_PACIENT_FK REFERENCES PACIENT(COD_PACIENT) ON DELETE CASCADE,
	ID_MILITAR INT
                	CONSTRAINT PROGRAMARE_ID_MILITAR_FK REFERENCES PERS_MIL(ID_MILITAR) ON DELETE CASCADE,
	DATA DATE
                   	CONSTRAINT PROGRAMARE_DATA_NN NOT NULL
);


--ANALIZE

CREATE SEQUENCE ANALIZE_COD_ANALIZA_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE ANALIZE(
	COD_ANALIZA INT
                	CONSTRAINT ANALIZE_COD_ANALIZA_PK PRIMARY KEY,
	COD_PACIENT INT
                	CONSTRAINT ANALIZE_COD_PACIENT_FK REFERENCES PACIENT(COD_PACIENT) ON DELETE CASCADE,
	DENUMIRE VARCHAR2(25)
                	CONSTRAINT ANALIZE_DENUMIRE_NN NOT NULL ,
	COST NUMBER(7,2),
	DATA_ANALIZA DATE
                	CONSTRAINT ANALIZE_DATA_ANALIZE_NN NOT NULL
);

--LABORATOR

CREATE SEQUENCE LABORATOR_ID_LAB_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE LABORATOR(
	ID_LAB INT
                	CONSTRAINT LABORATOR_ID_LAB_PK PRIMARY KEY,
	COD_ANALIZA INT
                	CONSTRAINT LABORATOR_COD_ANALIZA_FK REFERENCES ANALIZE(COD_ANALIZA) ON DELETE CASCADE,
 	NUME_LAB VARCHAR2(25)
                  	CONSTRAINT LABORATOR_NUME_LAB_NN NOT NULL
);


--REZULTATE ANALIZE

CREATE SEQUENCE REZULTAT_ID_REZULTAT_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE REZULTAT(
	ID_REZULTAT INT
                	CONSTRAINT REZULTAT_ID_REZULTAT_PK PRIMARY KEY,
	COD_ANALIZA INT
                	CONSTRAINT REZULTAT_COD_ANALIZA_FK REFERENCES ANALIZE(COD_ANALIZA) ON DELETE CASCADE,
	STATUS VARCHAR2(15)
                          	CONSTRAINT REZULTAT_STATUS_NN NOT NULL
);


--TRATAMENT

CREATE SEQUENCE TRATAMENT_COD_TRAT_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE TRATAMENT (
	COD_TRAT INT CONSTRAINT TRATAMENT_COD_TRAT_PK PRIMARY KEY,
	COD_PACIENT INT CONSTRAINT TRATAMENT_COD_PACIENT_FK REFERENCES PACIENT(COD_PACIENT) ON DELETE CASCADE,
    ID_REZULTAT INT CONSTRAINT TRATAMENT_ID_REZULTAT_FK REFERENCES REZULTAT(ID_REZULTAT) ON DELETE CASCADE
);


--PROIECT CERCETARE

CREATE SEQUENCE CERCETARE_ID_PROIECT_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE CERCETARE(
	ID_PROIECT INT
                	CONSTRAINT CERCETARE_ID_PROIECT_PK PRIMARY KEY,
	DESCRIERE VARCHAR2(40)
);


--MEDICAMENT
CREATE SEQUENCE MEDICAMENT_ID_MED_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE MEDICAMENT(
	ID_MED INT CONSTRAINT MEDICAMENT_ID_MED_PK PRIMARY KEY,
	DENUMIRE_MED VARCHAR2(25) CONSTRAINT MEDICAMENT_DENUMIRE_MED_NN NOT NULL,
	PRET NUMBER(7,2)
);


----------------------TABELE ASOCIATIVE--------------------
-- ANALIZE_PERSONAL

CREATE TABLE ANALIZE_PERSONAL (
ID_MILITAR INT REFERENCES PERS_MIL (ID_MILITAR), COD_ANALIZA INT REFERENCES ANALIZE (COD_ANALIZA), PRIMARY KEY (ID_MILITAR, COD_ANALIZA)
);

-- ANALIZE_CERCETARE
CREATE TABLE ANALIZE_CERCETARE (
COD_ANALIZA INT REFERENCES ANALIZE (COD_ANALIZA), ID_PROIECT INT REFERENCES CERCETARE (ID_PROIECT), PRIMARY KEY (COD_ANALIZA, ID_PROIECT)
);

-- MEDICAMENT_TRATAMENT
CREATE TABLE MEDICAMENT_TRATAMENT (
ID_MED INT REFERENCES MEDICAMENT (ID_MED), COD_TRAT INT REFERENCES TRATAMENT (COD_TRAT), PRIMARY KEY (ID_MED, COD_TRAT)
);

CREATE OR REPLACE PROCEDURE subprogram6 AS
   TYPE tip_analiza IS TABLE OF ANALIZE.DENUMIRE%TYPE INDEX BY PLS_INTEGER;
   analiza tip_analiza; -- table by index


   TYPE n_pers_mil IS TABLE OF PERS_MIL.ID_MILITAR%TYPE;
   personal n_pers_mil := n_pers_mil(); -- nested table


   TYPE pacienti IS TABLE OF VARCHAR2(200) INDEX BY PLS_INTEGER;
   nume_pacient pacienti; -- vector


   nr_pacienti NUMBER := 0;
   nr_analize NUMBER := 0;
   nr_personal NUMBER := 0;


BEGIN
   SELECT DISTINCT A.DENUMIRE
   BULK COLLECT INTO analiza
   FROM ANALIZE A; -- populam table by index (analiza)


   FOR i IN 1..analiza.COUNT LOOP -- parcurgem table by index (analiza)
       nr_personal := 0; -- resetam numarul de personal militar pentru fiecare analiza
       nr_pacienti := 0; -- resetam numarul de pacienti pentru fiecare analiza


       SELECT DISTINCT P.ID_MILITAR -- populam nested table (personal)
       BULK COLLECT INTO personal
       FROM PERS_MIL P
       JOIN ANALIZE_PERSONAL AP ON P.ID_MILITAR = AP.ID_MILITAR
       JOIN ANALIZE A ON AP.COD_ANALIZA = A.COD_ANALIZA
       WHERE A.DENUMIRE = analiza(i); -- analiza(i) este denumirea analizei curente


       SELECT DISTINCT P.NUME || ' ' || P.PRENUME -- populam vectorul (nume_pacient)
       BULK COLLECT INTO nume_pacient
       FROM PACIENT P
       JOIN ANALIZE A ON P.COD_PACIENT = A.COD_PACIENT
       WHERE A.DENUMIRE = analiza(i);


       nr_analize := nr_analize + 1;
       DBMS_OUTPUT.PUT_LINE('Analiza: ' || analiza(i));
       DBMS_OUTPUT.PUT_LINE(' ');


       FOR j IN 1..personal.COUNT LOOP
           nr_personal := nr_personal + 1;
           DBMS_OUTPUT.PUT_LINE('Militarul cu ID-ul ' || personal(j) || ' se ocupa de aceasta analiza.');
           DBMS_OUTPUT.PUT_LINE(' ');
       END LOOP;


       IF nr_personal = 0 THEN
           DBMS_OUTPUT.PUT_LINE('Nu exista personal militar care sa se ocupe de aceasta analiza.');
       END IF;


       FOR k IN 1..nume_pacient.COUNT LOOP
           nr_pacienti := nr_pacienti + 1;
           DBMS_OUTPUT.PUT_LINE('Pacient ' || nr_pacienti || ': ' || nume_pacient(k));
       END LOOP;


       IF nr_pacienti = 0 THEN
           DBMS_OUTPUT.PUT_LINE('NU EXISTA PACIENT!');
       END IF;
           DBMS_OUTPUT.PUT_LINE(' ');
           DBMS_OUTPUT.PUT_LINE(' ');
   END LOOP;


   IF nr_analize = 0 THEN
       DBMS_OUTPUT.PUT_LINE('Nu exista analize.');
   END IF;


END;
/


-- apelul subprogramului


BEGIN
   subprogram6;
END;

CREATE OR REPLACE PROCEDURE subprogram7 IS
   -- cursorul principal care contine informatii despre departamente
   CURSOR c_departamente IS
       SELECT ID_CATEGORIE, DENUMIRE
       FROM CATEGORIE;


   -- cursor parametrizat dependent de curs_departamente
   CURSOR c_militari (id_categorie_param NUMBER) IS -- parametrizat dupa id_categorie (departament)
       SELECT ID_MILITAR, NUME, PRENUME, SALARIU
       FROM PERS_MIL
       WHERE ID_CATEGORIE = id_categorie_param AND SALARIU > 7000; -- criteriu - salariu mai mare decat 7000


   -- variabile pentru stocarea datelor returnate de cursori
   v_id_categorie CATEGORIE.ID_CATEGORIE%TYPE;
   v_denumire CATEGORIE.DENUMIRE%TYPE;
   v_id_militar PERS_MIL.ID_MILITAR%TYPE;
   v_nume PERS_MIL.NUME%TYPE;
   v_prenume PERS_MIL.PRENUME%TYPE;
   v_salariu PERS_MIL.SALARIU%TYPE;


   cnt_departamente NUMBER := 0;
   cnt_pers_mil NUMBER := 0;


BEGIN
   -- descidere cursor
   OPEN c_departamente;


   -- iterare prin fiecare rand al cursorului principal
   LOOP
       cnt_pers_mil := 0; -- resetare numar personal militar pentru fiecare departament
       FETCH c_departamente INTO v_id_categorie, v_denumire;
       EXIT WHEN c_departamente%NOTFOUND; -- iesire din bucla cand cursorul principal nu mai are randuri


       -- afisare informatii despre departament
       cnt_departamente := cnt_departamente + 1;
       DBMS_OUTPUT.PUT_LINE('Departamentul ' || cnt_departamente || ': ' || v_denumire);
       DBMS_OUTPUT.PUT_LINE(' ');




       -- deschidere cursor parametrizat pentru militarii din departamentul curent
       OPEN c_militari(v_id_categorie);


       -- iterare prin fiecare rand al cursorului parametrizat
       LOOP
           FETCH c_militari INTO v_id_militar, v_nume, v_prenume, v_salariu;
           EXIT WHEN c_militari%NOTFOUND; -- iesire din bucla cand cursorul parametrizat nu mai are randuri


           -- afisare
           cnt_pers_mil := cnt_pers_mil + 1;
           DBMS_OUTPUT.PUT_LINE( ' ' ||cnt_pers_mil || '.' ||' Angajat: ' || v_nume || ' ' || v_prenume || ', Salariu: ' || v_salariu);


       END LOOP;


       IF(cnt_pers_mil = 0) THEN
           DBMS_OUTPUT.PUT_LINE('Nu exista personal militar in acest departament.');
       END IF;


       DBMS_OUTPUT.PUT_LINE(' ');
       DBMS_OUTPUT.PUT_LINE(' ');


       CLOSE c_militari;


   END LOOP;


   IF(cnt_departamente = 0) THEN
       DBMS_OUTPUT.PUT_LINE('Nu exista departamente.');
   END IF;


   CLOSE c_departamente;
END;
/


-- apelare subprogram
BEGIN
   subprogram7;
END;
/

CREATE OR REPLACE PROCEDURE procedura9 (
 p_id_proiect IN ANALIZE_CERCETARE.ID_PROIECT%TYPE,
 p_data_inceput IN ANALIZE.DATA_ANALIZA%TYPE,
 p_data_sfarsit IN ANALIZE.DATA_ANALIZA%TYPE
) IS
 -- variabile pentru gestionarea exceptiilor
 v_cnt_cercetare NUMBER;  
 v_cnt_analize NUMBER;


 -- exceptii personalizate
 NO_ID_PROIECT EXCEPTION;
 NO_DATA_FOUND_ANALIZE EXCEPTION;
 DATA_INCEPUT_INVALIDA EXCEPTION;
 DATA_INVALIDA EXCEPTION;


BEGIN
 -- verificam dacă id_proiect exista in tabela CERCETARE
 SELECT COUNT(*)
 INTO v_cnt_cercetare
 FROM CERCETARE
 WHERE ID_PROIECT = p_id_proiect;


 -- daca id_proiect nu exista, ridicam exceptia NO_ID_PROIECT
 IF v_cnt_cercetare = 0 THEN
   RAISE NO_ID_PROIECT;
 END IF;


 -- daca nu am analize, ridicam exceptia NO_ANALIZE
   IF v_cnt_analize = 0 THEN
   RAISE NO_DATA_FOUND_ANALIZE;
 END IF;


 -- daca datele sunt valide
 IF p_data_inceput IS NULL OR p_data_sfarsit IS NULL THEN
   RAISE_APPLICATION_ERROR(-20001, 'DATELE NU POT FI NULE!');
 END IF;


 IF p_data_inceput > p_data_sfarsit THEN
   RAISE DATA_INCEPUT_INVALIDA;
 END IF;




 -- interogare pentru a obtine analizele, rezultatele si tratamentele asociate
 SELECT COUNT(*)
 INTO v_cnt_analize
 FROM ANALIZE A
 JOIN ANALIZE_CERCETARE AC ON A.COD_ANALIZA = AC.COD_ANALIZA
 JOIN CERCETARE C ON AC.ID_PROIECT = C.ID_PROIECT
 JOIN REZULTAT R ON A.COD_ANALIZA = R.COD_ANALIZA
 JOIN TRATAMENT T ON R.ID_REZULTAT = T.ID_REZULTAT
 WHERE C.ID_PROIECT = p_id_proiect
   AND A.DATA_ANALIZA BETWEEN p_data_inceput AND p_data_sfarsit;


 IF v_cnt_analize = 0 THEN
   RAISE_APPLICATION_ERROR(-20001, 'NU EXISTA INREGISTRARI PENTRU ACEASTA PERIOADA!');
   RETURN; -- iesire din procedura daca nu exista inregistrari
 END IF;


-- interogare pentru a obtine analizele, rezultatele si tratamentele asociate
   FOR reg IN (
     SELECT A.COD_ANALIZA, R.STATUS, T.COD_TRAT
     FROM ANALIZE A -- tabel 1
     JOIN ANALIZE_CERCETARE AC ON A.COD_ANALIZA = AC.COD_ANALIZA -- tabel 2
     JOIN CERCETARE C ON AC.ID_PROIECT = C.ID_PROIECT -- tabel 3
     JOIN REZULTAT R ON A.COD_ANALIZA = R.COD_ANALIZA -- tabel 4
     JOIN TRATAMENT T ON R.ID_REZULTAT = T.ID_REZULTAT -- tabel 5
     WHERE C.ID_PROIECT = p_id_proiect
       AND A.DATA_ANALIZA BETWEEN p_data_inceput AND p_data_sfarsit
   ) LOOP
     DBMS_OUTPUT.PUT_LINE('COD ANALIZA: ' || reg.COD_ANALIZA || ', STATUS REZULTAT: ' || reg.STATUS || ', COD TRATAMENT: ' || reg.COD_TRAT);
   END LOOP;




EXCEPTION
 WHEN NO_ID_PROIECT THEN
   DBMS_OUTPUT.PUT_LINE('NU EXISTA ACEST PROIECT DE CERCETARE CU ACEST ID! - ' || SQLERRM);


 WHEN DATA_INCEPUT_INVALIDA THEN
   DBMS_OUTPUT.PUT_LINE('DATA DE INCEPUT ESTE MAI MARE DECAT CEA DE SFARSIT! - ' || SQLERRM);


 WHEN NO_DATA_FOUND_ANALIZE THEN
     DBMS_OUTPUT.PUT_LINE('NU EXISTA INREGISTRARI PENTRU ACEASTA PERIOADA! - ' || SQLERRM);


 WHEN OTHERS THEN
   DBMS_OUTPUT.PUT_LINE('A INTERVENIT O EROARE:  - ' || SQLERRM);


END procedura9;
/
