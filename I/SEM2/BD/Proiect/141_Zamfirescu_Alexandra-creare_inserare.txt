-------------------CREATE TABLE-----------------------

--PACIENT

CREATE SEQUENCE PACIENT_ID_PACIENT_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE PACIENT(
                        ID_PACIENT INT
                            CONSTRAINT PACIENT_ID_PACIENT_PK PRIMARY KEY,
                        NUME VARCHAR2(25)
                            CONSTRAINT PACIENT_NUME_NN NOT NULL,
                        PRENUME VARCHAR2(30)
                            CONSTRAINT PACIENT_PRENUME_NN NOT NULL,
                        SEX VARCHAR2(10)
                            CONSTRAINT PACIENT_SEX_NN NOT NULL,
                        CNP VARCHAR2(20)
                            CONSTRAINT PACIENT_CNP_NN NOT NULL,
                        NR_TELEFON VARCHAR2(15),
                        ADRESA VARCHAR2(40)
);


--DEPARTAMENT

CREATE SEQUENCE DEPARTAMENT_ID_DEPARTAMENT_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE DEPARTAMENT(
    ID_DEPARTAMENT INT
                    CONSTRAINT DEPARTAMENT_ID_DEPARTAMENT_PK PRIMARY KEY,
    DENUMIRE VARCHAR2(30)
                    CONSTRAINT DEPARTAMENT_DENUMIRE_NN NOT NULL
);

--TRATAMENT

CREATE SEQUENCE TRATAMENT_ID_TRATAMENT_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE TRATAMENT(
    ID_TRATAMENT INT
                    CONSTRAINT TRATAMENT_ID_TRATAMENT_PK PRIMARY KEY,
     ID_PACIENT INT
                    CONSTRAINT TRATAMENT_ID_PACIENT_FK REFERENCES PACIENT(ID_PACIENT) ON DELETE CASCADE
);


--LABORATOR

CREATE SEQUENCE LABORATOR_ID_LABORATOR_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE LABORATOR(
    ID_LABORATOR INT
                    CONSTRAINT LABORATOR_ID_LABORATOR_PK PRIMARY KEY,
     NUME_LAB VARCHAR2(25)
                      CONSTRAINT LABORATOR_NUME_LAB_NN NOT NULL
);


--PERSONAL MILITAR

CREATE SEQUENCE PERSONAL_MILITAR_ID_PERSONAL_MILITAR_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE PERSONAL_MILITAR(
    ID_PERSONAL_MILITAR INT
                    CONSTRAINT PERSONAL_MILITAR_ID_PERSONAL_MILITAR_PK PRIMARY KEY,
    ID_DEPARTAMENT INT
                    CONSTRAINT PERSONAL_MILITAR_ID_DEPARTAMENT_FK REFERENCES DEPARTAMENT(ID_DEPARTAMENT) ON DELETE CASCADE,
    NUME VARCHAR2(25)
                    CONSTRAINT PERSONAL_MILITAR_NUME_NN NOT NULL,
    PRENUME VARCHAR2(30)
                    CONSTRAINT PERSONAL_MILITAR_PRENUME_NN NOT NULL,
    GRAD_MILITAR VARCHAR2(15)
                             CONSTRAINT PERSONAL_MILITAR_GRAD_MILITAR_NN NOT NULL,
    SALARIU NUMBER(8,2)
                             CONSTRAINT PERSONAL_MILITAR_SALARIU_CK CHECK ( SALARIU >= 5000 ),
    DATA_ANGAJARE DATE,
    EMAIL VARCHAR2(25)
);


--PROGRAMARE

CREATE SEQUENCE PROGRAMARE_ID_PROGRAMARE_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE PROGRAMARE(
    ID_PROGRAMARE INT
                    CONSTRAINT PROGRAMARE_ID_PROGRAMARE_PK PRIMARY KEY,
    ID_PACIENT INT
                    CONSTRAINT PROGRAMARE_ID_PACIENT_FK REFERENCES PACIENT(ID_PACIENT) ON DELETE CASCADE,
    ID_PERSONAL_MILITAR INT
                    CONSTRAINT PROGRAMARE_ID_PERSONAL_MILITAR_FK REFERENCES PERSONAL_MILITAR(ID_PERSONAL_MILITAR) ON DELETE CASCADE,
    DATA DATE
                       CONSTRAINT PROGRAMARE_DATA_NN NOT NULL
);


--ANALIZE

CREATE SEQUENCE ANALIZE_ID_ANALIZE_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE ANALIZE(
    ID_ANALIZE INT
                    CONSTRAINT ANALIZE_ID_ANALIZE_PK PRIMARY KEY,
    ID_PACIENT INT
                    CONSTRAINT ANALIZE_ID_PACIENT_FK REFERENCES PACIENT(ID_PACIENT) ON DELETE CASCADE,
    ID_PERSONAL_MILITAR INT
                    CONSTRAINT ANALIZE_ID_PERSONAL_MILITAR_FK REFERENCES PERSONAL_MILITAR(ID_PERSONAL_MILITAR) ON DELETE CASCADE,
    ID_LABORATOR INT
                    CONSTRAINT ANALIZE_ID_LABORATOR_FK REFERENCES LABORATOR(ID_LABORATOR) ON DELETE CASCADE,
    DENUMIRE VARCHAR2(25)
                    CONSTRAINT ANALIZE_DENUMIRE_NN NOT NULL ,
    COST NUMBER(7,2),
    DATA_ANALIZA DATE
                    CONSTRAINT ANALIZE_DATA_ANALIZE_NN NOT NULL
);


--REZULTATE ANALIZE

CREATE SEQUENCE REZULTATE_ANALIZE_ID_REZULTAT_SEQ
START WITH 1
MAXVALUE 9999999
NOCACHE NOCYCLE;


CREATE TABLE REZULTATE_ANALIZE(
    ID_REZULTAT INT
                    CONSTRAINT REZULTATE_ANALIZE_ID_REZULTAT_PK PRIMARY KEY,
    ID_TRATAMENT INT

    ID_PERSONAL_MILITAR INT
                    CONSTRAINT REZULTATE_ANALIZE_ID_PERSONAL_MILITAR_FK REFERENCES PERSONAL_MILITAR(ID_PERSONAL_MILITAR) ON DELETE CASCADE,
    STATUS VARCHAR2(15)
                              CONSTRAINT REZULTATE_ANALIZE_STATUS_NN NOT NULL
);

NOCACHE NOCYCLE;
